portable_rust (Online TW) 啟動 startup.bat:

portable rust (E Water)

@echo off
chcp 65001 >nul
setlocal ENABLEEXTENSIONS

REM ==================================================
REM Portable System Bootstrap (Rust)
REM ==================================================

REM === 虛擬磁碟代號（請勿加反斜線）===
set DRIVE=R:

REM === for Python312
set PYTHONPATH=%DRIVE%\Python312\DLLs;%DRIVE%\Python312\Lib;%DRIVE%\Python312\Lib\site-packages
set PYTHONIOENCODING=utf-8

REM === 取得 batch 所在目錄（結尾含 \）===
set ROOT=%~dp0
set DATA=%ROOT%data

REM === 若已存在則先解除（避免錯誤）===
subst %DRIVE% /D >nul 2>&1

REM === 建立虛擬磁碟 ===
subst %DRIVE% "%DATA%"

REM ==================================================
REM Home / User 環境
REM ==================================================

set HOMEDRIVE=%DRIVE%
set HOMEPATH=\home
set USERPROFILE=%DRIVE%\home
set HOME=%DRIVE%\home

REM ==================================================
REM Git / Putty
REM ==================================================

set GIT_SSH=%DRIVE%\PuTTY\plink.exe

REM ==================================================
REM Rust (Portable, GNU)
REM ==================================================

set RUSTUP_HOME=%DRIVE%\.rustup
REM set CARGO_HOME=%DRIVE%\.cargo
set CARGO_HOME=C:\cargo
set SCITE_HOME=%DRIVE%\wscite
set RUSTUP_NO_UPDATE_CHECK=1

REM === 強制使用 GNU toolchain ===
set RUSTUP_TOOLCHAIN=stable-x86_64-pc-windows-gnu

REM === MinGW 編譯器指定 ===
set CC=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX=%DRIVE%\msys64\ucrt64\bin\g++.exe
set AR=%DRIVE%\msys64\ucrt64\bin\ar.exe

REM === Rust linker 指定 ===
set RUSTFLAGS=-C linker=%DRIVE%\msys64\ucrt64\bin\gcc.exe

REM ==================================================
REM MSYS2
REM ==================================================

set MSYS2=%DRIVE%\msys64
set CC=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX=%DRIVE%\msys64\ucrt64\bin\g++.exe
set CC_x86_64_pc_windows_gnu=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX_x86_64_pc_windows_gnu=%DRIVE%\msys64\ucrt64\bin\g++.exe

REM ==================================================
REM PATH 組合
REM ==================================================

set PATH=%DRIVE%\msys64\ucrt64\bin;%DRIVE%\msys64\usr\bin;%CARGO_HOME%\bin;%SCITE_HOME%;%DRIVE%\portablegit\bin;%DRIVE%\Zed;%PATH%

set PATH=%DRIVE%\Python312;%DRIVE%\Python312\Scripts;%PATH%;%DRIVE%;%DRIVE%\.rustup\toolchains\stable-x86_64-pc-windows-gnu\bin

REM ==================================================
REM 顯示資訊
REM ==================================================

title Portable Rust GNU Environment (%DRIVE%)

echo ======================================
echo  Portable Environment Ready
echo  Mounted Drive : %DRIVE%
echo  Data Path     : %DATA%
echo  Rust Toolchain: %RUSTUP_TOOLCHAIN%
echo  MinGW GCC     : %CC%
echo  Type STOP then close window to exit
echo ======================================
echo.

REM ==================================================
REM 啟動工具
REM ==================================================

if exist "%SCITE_HOME%\SciTE.exe" (
    start "" /MIN "%SCITE_HOME%\SciTE.exe"
    start "" /MIN "%SCITE_HOME%\SciTE.exe"
)

REM === 切換到虛擬磁碟 ===

%DRIVE%

REM ==================================================
REM 保持命令列
REM ==================================================

start /min "CMD-1" cmd /k
start /min "CMD-2" cmd /k
start /min "CMD-3" cmd /k
start /min "CMD-4" cmd /k

REM ==================================================
REM 啟動 rustcms
REM ==================================================

start /min "rustcms" cmd /k "cd /d %DRIVE%\projects\rustcms && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5000

REM ==================================================
REM 啟動 all
REM ==================================================

start /min "all" cmd /k "cd /d %DRIVE%\projects\cycuorg\all && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5001

REM ==================================================
REM 啟動 art
REM ==================================================

start /min "art" cmd /k "cd /d %DRIVE%\projects\cycuorg\art && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5002

REM ==================================================
REM 啟動 clothing
REM ==================================================

echo.
echo ======================================
echo Starting CLOTHING CMS
echo ======================================

REM 確認原始目錄

if not exist "%DRIVE%\projects\cycuorg\clothing\cms.exe" (
    echo ERROR: clothing cms.exe not found.
    goto CLOTHING_FIREFOX
)

REM 在原始目錄啟動，不移動、不複製

start /min "clothing" cmd /k "cd /d %DRIVE%\projects\cycuorg\clothing && cms.exe"

REM 等待 Port 5003，最多 30 秒

set COUNT=0

:WAIT_CLOTHING

powershell -NoProfile -Command ^
"$c = New-Object System.Net.Sockets.TcpClient; try {$c.Connect('127.0.0.1',5003); exit 0} catch {exit 1} finally {$c.Dispose()}" >nul 2>&1

if not errorlevel 1 goto CLOTHING_READY

set /a COUNT+=1

if %COUNT% GEQ 30 goto CLOTHING_TIMEOUT

timeout /t 1 /nobreak >nul

goto WAIT_CLOTHING


:CLOTHING_READY

echo.
echo [OK] CLOTHING HTTPS PORT 5003 READY.

goto CLOTHING_FIREFOX


:CLOTHING_TIMEOUT

echo.
echo [WARNING] CLOTHING PORT 5003 NOT READY.
echo Please check clothing CMD window.
echo.

:CLOTHING_FIREFOX

REM 無論是否監聽成功，都保留 Firefox 開啟動作

start "" firefox -new-tab https://localhost:5003

REM ==================================================
REM 啟動 health
REM ==================================================

start /min "health" cmd /k "cd /d %DRIVE%\projects\cycuorg\health && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5004

REM ==================================================
REM 啟動 history
REM ==================================================

start /min "history" cmd /k "cd /d %DRIVE%\projects\cycuorg\history && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5005

REM ==================================================
REM 啟動 learning
REM ==================================================

start /min "learning" cmd /k "cd /d %DRIVE%\projects\cycuorg\learning && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5006

REM ==================================================
REM 啟動 living
REM ==================================================

start /min "living" cmd /k "cd /d %DRIVE%\projects\cycuorg\living && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5007

REM ==================================================
REM 啟動 recipes
REM ==================================================

start /min "recipes" cmd /k "cd /d %DRIVE%\projects\cycuorg\recipes && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5008

REM ==================================================
REM 啟動 tech
REM ==================================================

start /min "tech" cmd /k "cd /d %DRIVE%\projects\cycuorg\tech && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5009

REM ==================================================
REM 啟動 tourism
REM ==================================================

start /min "tourism" cmd /k "cd /d %DRIVE%\projects\cycuorg\tourism && cms.exe"

timeout /t 3 /nobreak >nul

start "" firefox -new-tab https://localhost:5010

REM ==================================================
REM 啟動 Zed
REM ==================================================

Zed %DRIVE%\projects\rustcms

REM ==================================================
REM 結束
REM ==================================================

echo.
echo ======================================
echo  ALL CMS STARTUP COMMANDS COMPLETED
echo ======================================

REM 保持原本批次視窗

關閉 stop.bat:

@echo off
set DRIVE=R:

taskkill /IM python.exe /F
taskkill /IM pythonw.exe /F
REM taskkill /IM adb.exe /F
taskkill /IM scite.exe /F
taskkill /IM ugraf.exe /F
taskkill /IM webotsw.exe /F
taskkill /IM solvespace.exe /F
taskkill /IM sharex.exe /F
taskkill /IM leo.exe /F
taskkill /IM zed.exe /F
taskkill /IM cms.exe /F

echo Unmounting %DRIVE% ...
subst %DRIVE% /D

REM 關閉 cmd 指令視窗
taskkill /IM cmd.exe /F

echo Done.

EXIT
