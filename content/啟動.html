portable_rust 啟動 startup.bat:

@echo off
chcp 65001 >nul
setlocal ENABLEEXTENSIONS

REM ==================================================
REM  Portable System Bootstrap (Rust)
REM ==================================================

REM === 虛擬磁碟代號（請勿加反斜線）===
set DRIVE=R:

REM === for Python312
set PYTHONPATH=%DRIVE%\Python312\DLLs;%DRIVE%\Python312\Lib;%DRIVE%\Python312\Lib\site-packages;
REM 將系統 Python 程式的 io 設為 utf-8
set PYTHONIOENCODING="utf-8"

REM === 取得 batch 所在目錄（結尾含 \）===
set ROOT=%~dp0
set DATA=%ROOT%data

REM === 若已存在則先解除（避免錯誤）===
subst %DRIVE% /D >nul 2>&1

REM === 建立虛擬磁碟 ===
subst %DRIVE% "%DATA%"

REM ==================================================
REM  Home / User 環境（避免污染系統）
REM ==================================================
set HOMEDRIVE=%DRIVE%
set HOMEPATH=\home
set USERPROFILE=%DRIVE%\home
set HOME=%DRIVE%\home

REM ==================================================
REM  Git / Putty
REM ==================================================
set GIT_SSH=%DRIVE%\PuTTY\plink.exe

REM ==================================================
REM  Rust (Portable, GNU)
REM ==================================================
set RUSTUP_HOME=%DRIVE%\.rustup
REM et CARGO_HOME=%DRIVE%\.cargo
REM cargo cache need NTFS disk
set CARGO_HOME=C:\cargo
set SCITE_HOME=%DRIVE%\wscite
set RUSTUP_NO_UPDATE_CHECK=1

REM === 強制使用 GNU toolchain ===
set RUSTUP_TOOLCHAIN=stable-x86_64-pc-windows-gnu

REM === MinGW 編譯器指定（非常重要）===
set CC=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX=%DRIVE%\msys64\ucrt64\bin\g++.exe
set AR=%DRIVE%\msys64\ucrt64\bin\ar.exe

REM === Rust linker 指定 ===
set RUSTFLAGS=-C linker=%DRIVE%\msys64\ucrt64\bin\gcc.exe

REM === unveil-rs ===
set MSYS2=%DRIVE%\msys64
set CC=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX=%DRIVE%\msys64\ucrt64\bin\g++.exe
set CC_x86_64_pc_windows_gnu=%DRIVE%\msys64\ucrt64\bin\gcc.exe
set CXX_x86_64_pc_windows_gnu=%DRIVE%\msys64\ucrt64\bin\g++.exe

REM ==================================================
REM  PATH 組合
REM ==================================================
set PATH=%DRIVE%\msys64\ucrt64\bin^
;%DRIVE%\msys64\usr\bin^
;%CARGO_HOME%\bin^
;%SCITE_HOME%^
;%DRIVE%\portablegit\bin^
;%DRIVE%\Zed
;%PATH%

REM 以可攜系統中的 Python312 為主
set PATH=%DRIVE%\Python312;%DRIVE%\Python312\Scripts;%PATH%;%DRIVE%;%DRIVE%\.rustup\toolchains\stable-x86_64-pc-windows-gnu\bin;

REM ==================================================
REM  顯示資訊
REM ==================================================
title Portable Rust GNU Environment (%DRIVE%)

echo ======================================
echo  Portable Environment Ready
echo  Mounted Drive : %DRIVE%
echo  Data Path     : %DATA%
echo  Rust Toolchain: %RUSTUP_TOOLCHAIN%
echo  MinGW GCC     : %CC%
echo  Type STOP then close window to exit
echo ======================================
echo.

REM ==================================================
REM  啟動工具
REM ==================================================
if exist "%SCITE_HOME%\SciTE.exe" (
    start "" /MIN "%SCITE_HOME%\SciTE.exe"
    start "" /MIN "%SCITE_HOME%\SciTE.exe"
)

REM === 切換到虛擬磁碟 ===
%DRIVE%

REM ==================================================
REM  保持命令列
REM ==================================================
start /min "CMD-1" cmd /k
start /min  "CMD-2" cmd /k
start /min" CMD-3" cmd /k
start /min "CMD-4" cmd /k

REM　啟動 run-rustcms.exe 之後用 firefox 開啟動態網頁編輯
start /min "rustcms" cmd /k "cd /d %DRIVE%\projects\rustcms && cms.exe"
timeout /t 3 /nobreak >nul
start "" firefox -new-tab https://localhost:5000

REM　啟動 all
start /min "all" cmd /k "cd /d %DRIVE%\projects\cycuorg\all && cms.exe"
timeout /t 3 /nobreak >nul
start "" firefox -new-tab https://localhost:5001

REM　啟動 art
start /min "art" cmd /k "cd /d %DRIVE%\projects\cycuorg\art && cms.exe"
timeout /t 3 /nobreak >nul
start "" firefox -new-tab https://localhost:5002

REM　啟動 clothing
start /min "clothing" cmd /k "cd /d %DRIVE%\projects\cycuorg\clothing && cms.exe"
timeout /t 3 /nobreak >nul
start "" firefox -new-tab https://localhost:5003

關閉 stop.bat:

@echo off
set DRIVE=R:

taskkill /IM python.exe /F
taskkill /IM pythonw.exe /F
REM taskkill /IM adb.exe /F
taskkill /IM scite.exe /F
taskkill /IM ugraf.exe /F
taskkill /IM webotsw.exe /F
taskkill /IM solvespace.exe /F
taskkill /IM sharex.exe /F
taskkill /IM leo.exe /F
taskkill /IM zed.exe /F
taskkill /IM cms.exe /F

echo Unmounting %DRIVE% ...
subst %DRIVE% /D

REM 關閉 cmd 指令視窗
taskkill /IM cmd.exe /F

echo Done.

EXIT
